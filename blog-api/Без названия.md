● Команды для установки Sanctum                                                                                                                                                                                                                         
  # 1. Установка Sanctum
  composer require laravel/sanctum                                                                                                                                  
  # 2. Публикация конфигурации и миграций                                                                                                                        
  php artisan vendor:publish --provider="Laravel\Sanctum\SanctumServiceProvider"

  # 3. Запуск миграций (создаст personal_access_tokens)
  php artisan migrate

  # 4. Добавить HasApiTokens в User модель (app/Models/User.php)

  В app/Models/User.php добавить:
  use Laravel\Sanctum\HasApiTokens;

  class User extends Model
  {
      use HasApiTokens, HasFactory, Notifiable;
      // ...
  }

  Также потребуется создать

  # 5. Создать routes/api.php
  touch routes/api.php

  # 6. Зарегистрировать routes в bootstrap/app.php

  В bootstrap/app.php добавить:
  ->withRouting(
      api: __DIR__.'/../routes/api.php',
      // ...
  )
